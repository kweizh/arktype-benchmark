// Simple resolve
function resolveRefs(schema: any, defs?: any, stack: Set<string> = new Set()): any {
  if (typeof schema !== "object" || schema === null) return schema;
  if (Array.isArray(schema)) return schema.map((s: any) => resolveRefs(s, defs, stack));
  
  const d = defs || schema.$defs;
  const resolved: any = {};
  for (const [key, value] of Object.entries(schema)) {
    if (key === "$defs") continue;
    if (key === "$ref" && d && typeof value === "string" && value.startsWith("#/$defs/")) {
      const defName = value.slice("#/$defs/".length);
      if (stack.has(defName)) {
        // Cycle detected - return true (accept anything)
        return true;
      }
      const defSchema = d[defName];
      if (defSchema) {
        stack.add(defName);
        const resolvedDef = resolveRefs(defSchema, d, stack);
        stack.delete(defName);
        Object.assign(resolved, resolvedDef);
        continue;
      }
    }
    resolved[key] = resolveRefs(value, d, stack);
  }
  return resolved;
}
