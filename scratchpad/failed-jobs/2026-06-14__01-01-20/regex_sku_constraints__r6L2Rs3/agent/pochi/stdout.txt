import('arktype').then(m => {
  const skuType = m.type('/^[A-Z]{2}-\\\\d{4}-[a-z]{3}\$/');
  const skusType = skuType.array().narrow((skus, ctx) => {
    if (skus.length < 1 || skus.length > 200) return ctx.mustBe('between 1 and 200 elements');
    const seen = new Set();
    for (let i = 0; i < skus.length; i++) {
      if (seen.has(skus[i])) return ctx.mustBe('unique');
      seen.add(skus[i]);
    }
    return true;
  });
