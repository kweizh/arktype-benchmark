### `applyTax`
```ts
export const applyTax = type.fn(
  "number",        // param [0]: amount
  "number = 0.1", // param [1]: rate (optional, default 0.1)
  ":",
  "number"         // return type
)((amount, rate) => {
  return parseFloat((amount * (1 + rate)).toPrecision(15));
});
```

### `formatInvoice`
```ts
export const formatInvoice = type.fn(
  "number[]",    // param [0]: items
  "number = 0",  // param [1]: discount (optional, default 0)
  ":",
  { total: "number", count: "number" }  // return type
)((items, discount) => {
  const sum = items.reduce((acc, n) => acc + n, 0);
  return { total: sum * (1 - discount), count: items.length };
});
```
