  HOST: "string.ip | 'localhost'",        // IPv4/IPv6 or literal "localhost"
  PORT: "1 <= number.integer <= 65535",   // integer in [1..65535]
  DEBUG: "boolean",                       // "true"/"false" → boolean

});
