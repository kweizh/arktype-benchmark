```typescript
import { scope } from "arktype";

export const { Member } = scope({
  Member: {
    id: "string.uuid",
    name: "string",
    "manager?": "Member",
    "subordinates?": "Member[]",
  },
}).export();

export function findRoot(member: typeof Member.infer): typeof Member.infer {
  let current = member;
  while (current.manager) {
    current = current.manager;
  }
  return current;
}
```
